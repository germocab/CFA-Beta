-- ========================================================= -- MIGRACIÓN: Múltiples personas por turno + Horarios automáticos por punto -- Ejecutar en el SQL Editor de Supabase, en orden, de arriba hacia abajo. -- ========================================================= -- --------------------------------------------------------- -- 1) NUEVAS COLUMNAS EN "turnos": persona_ids (varias personas) -- --------------------------------------------------------- ALTER TABLE turnos ADD COLUMN IF NOT EXISTS persona_ids jsonb DEFAULT '[]'::jsonb; -- Migramos los datos viejos: si un turno ya tenía persona_id (single), -- lo convertimos en un arreglo de 1 elemento para no perder información. UPDATE turnos SET persona_ids = jsonb_build_array(persona_id) WHERE persona_id IS NOT NULL AND (persona_ids IS NULL OR persona_ids = '[]'::jsonb); -- (Opcional, puedes dejarla por compatibilidad o borrarla más adelante) -- ALTER TABLE turnos DROP COLUMN persona_id; -- --------------------------------------------------------- -- 2) NUEVAS COLUMNAS EN "puntos": horarios predeterminados por tipo de turno -- --------------------------------------------------------- ALTER TABLE puntos ADD COLUMN IF NOT EXISTS horario_m text DEFAULT ''; ALTER TABLE puntos ADD COLUMN IF NOT EXISTS horario_t text DEFAULT ''; ALTER TABLE puntos ADD COLUMN IF NOT EXISTS horario_td text DEFAULT ''; -- --------------------------------------------------------- -- 3) FUNCIÓN: admin_crear_turno (ahora recibe p_persona_ids en vez de p_persona_id) -- --------------------------------------------------------- DROP FUNCTION IF EXISTS admin_crear_turno(text, uuid, text, text, text, text, uuid, text); CREATE OR REPLACE FUNCTION admin_crear_turno( p_password text, p_punto_id uuid, p_dia text, p_tipo text, p_hora text, p_nombres text, p_persona_ids jsonb, p_estado text ) RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_id uuid; BEGIN IF NOT verificar_admin_password(p_password) THEN RAISE EXCEPTION 'Contraseña incorrecta'; END IF; INSERT INTO turnos (punto_id, dia, tipo, hora, nombres, persona_ids, estado) VALUES (p_punto_id, p_dia, p_tipo, p_hora, p_nombres, p_persona_ids, p_estado) RETURNING id INTO v_id; RETURN v_id; END; $$; -- --------------------------------------------------------- -- 4) FUNCIÓN: admin_editar_turno (ahora recibe p_persona_ids en vez de p_persona_id) -- --------------------------------------------------------- DROP FUNCTION IF EXISTS admin_editar_turno(text, uuid, uuid, text, text, text, text, uuid, text); CREATE OR REPLACE FUNCTION admin_editar_turno( p_password text, p_id uuid, p_punto_id uuid, p_dia text, p_tipo text, p_hora text, p_nombres text, p_persona_ids jsonb, p_estado text ) RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$ BEGIN IF NOT verificar_admin_password(p_password) THEN RAISE EXCEPTION 'Contraseña incorrecta'; END IF; UPDATE turnos SET punto_id = p_punto_id, dia = p_dia, tipo = p_tipo, hora = p_hora, nombres = p_nombres, persona_ids = p_persona_ids, estado = p_estado WHERE id = p_id; END; $$; -- --------------------------------------------------------- -- 5) FUNCIÓN: admin_crear_punto (agrega p_horario_m, p_horario_t, p_horario_td) -- --------------------------------------------------------- DROP FUNCTION IF EXISTS admin_crear_punto(text, text, text, text[], text, text, text, text, text, integer); CREATE OR REPLACE FUNCTION admin_crear_punto( p_password text, p_nombre text, p_color text, p_dias text[], p_retiro_nombre text, p_retiro_sub text, p_retiro_tel text, p_encargado_nombre text, p_encargado_tel text, p_orden integer, p_horario_m text, p_horario_t text, p_horario_td text ) RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_id uuid; BEGIN IF NOT verificar_admin_password(p_password) THEN RAISE EXCEPTION 'Contraseña incorrecta'; END IF; INSERT INTO puntos ( nombre, color, dias, retiro_nombre, retiro_sub, retiro_tel, encargado_nombre, encargado_tel, orden, horario_m, horario_t, horario_td ) VALUES ( p_nombre, p_color, p_dias, p_retiro_nombre, p_retiro_sub, p_retiro_tel, p_encargado_nombre, p_encargado_tel, p_orden, p_horario_m, p_horario_t, p_horario_td ) RETURNING id INTO v_id; RETURN v_id; END; $$; -- --------------------------------------------------------- -- 6) FUNCIÓN: admin_editar_punto (agrega p_horario_m, p_horario_t, p_horario_td) -- --------------------------------------------------------- DROP FUNCTION IF EXISTS admin_editar_punto(text, uuid, text, text, text[], text, text, text, text, text, integer); CREATE OR REPLACE FUNCTION admin_editar_punto( p_password text, p_id uuid, p_nombre text, p_color text, p_dias text[], p_retiro_nombre text, p_retiro_sub text, p_retiro_tel text, p_encargado_nombre text, p_encargado_tel text, p_orden integer, p_horario_m text, p_horario_t text, p_horario_td text ) RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$ BEGIN IF NOT verificar_admin_password(p_password) THEN RAISE EXCEPTION 'Contraseña incorrecta'; END IF; UPDATE puntos SET nombre = p_nombre, color = p_color, dias = p_dias, retiro_nombre = p_retiro_nombre, retiro_sub = p_retiro_sub, retiro_tel = p_retiro_tel, encargado_nombre = p_encargado_nombre, encargado_tel = p_encargado_tel, orden = p_orden, horario_m = p_horario_m, horario_t = p_horario_t, horario_td = p_horario_td WHERE id = p_id; END; $$; -- ========================================================= -- FIN DE LA MIGRACIÓN -- ========================================================= -- NOTA IMPORTANTE: Los nombres exactos de los parámetros (p_password, -- p_punto_id, etc.) y sus tipos (uuid, text, text[], jsonb...) deben -- coincidir con los que ya tenías en tu base. Si tu función original -- usaba un tipo distinto para algún parámetro (por ejemplo "id" en vez -- de "uuid"), revisa el DROP FUNCTION correspondiente: si no encuentra -- una función con esa firma exacta, no hace nada (no falla), por lo -- que el CREATE OR REPLACE de abajo simplemente crea una nueva versión -- y puede que quede una versión vieja "duplicada".
